// Toggle scroll limitations based on screen view status
function startApp(screenId) {
  document.body.classList.remove('no-scroll');
  switchScreen(screenId);
}

function switchScreen(screenId) {
  document.querySelectorAll('.app-screen').forEach(screen => {
    screen.classList.remove('active-screen');
  });
  document.getElementById(screenId).classList.add('active-screen');
  window.scrollTo(0, 0);
}

let selectedButton = null; 
let conversationIndices = []; 
let workingPool = []; 
let currentCardIndex = 0; 
let currentKeywords = []; 

function buildMenu() { 
  const menu = document.getElementById("topicMenu"); 
  menu.innerHTML = ""; 
  Object.keys(topics).forEach((title, idx) => { 
    const btn = document.createElement("button");
    btn.className = "topic-btn"; 
    btn.textContent = title; 
    if(idx === 0) { 
      btn.classList.add("active"); 
      selectedButton = btn;
    } 
    btn.onclick = (e) => handleTopicSwitch(e.target, title); 
    menu.appendChild(btn); 
  }); 
} 

function handleTopicSwitch(targetBtn, title) { 
  if(selectedButton) selectedButton.classList.remove("active"); 
  targetBtn.classList.add("active"); 
  selectedButton = targetBtn;
  loadTopic(title); 
} 

function loadTopic(title) { 
  const data = topics[title]; 
  workingPool = data.dialogue.map((text, index) => ({ text, originalIndex: index }));
  for (let i = workingPool.length - 1; i > 0; i--) { 
    const j = Math.floor(Math.random() * (i + 1));
    [workingPool[i], workingPool[j]] = [workingPool[j], workingPool[i]]; 
  } 
  currentKeywords = data.keywords; 
  currentCardIndex = 0; 
  renderKeywordsAndSample(data); 
  updateFlashcard(); 
  resetConversation();
} 

function renderKeywordsAndSample(data) { 
  const kwBox = document.getElementById("keywordBox"); 
  kwBox.innerHTML = "";
  data.keywords.forEach(kw => {
    const div = document.createElement("div");
    div.className = "keyword-tag";
    div.innerHTML = `${kw.word} <span>${kw.def}</span>`;
    kwBox.appendChild(div);
  });

  const sampleBox = document.getElementById("sampleConvBox");
  sampleBox.innerHTML = `<p style="font-weight:800; margin-bottom:12px; color:var(--primary-700);">Correct Sentence Sequence:</p>` + 
    data.dialogue.map(line => `<p style="margin: 6px 0;">${boldRoles(line, data.roles)}</p>`).join("");
}

function boldRoles(text, roles) {
  const roleValues = Object.values(roles);
  let targetKey = "";
  let roleClass = "";
  
  if (text.startsWith(roleValues[0] + ":")) {
    targetKey = roleValues[0] + ":";
    roleClass = "role-a";
  } else if (text.startsWith(roleValues[1] + ":")) {
    targetKey = roleValues[1] + ":";
    roleClass = "role-b";
  }

  if (targetKey) {
    return `<span class="role-tag ${roleClass}">${targetKey.replace(':','')}</span>` + text.substring(targetKey.length);
  }
  return text;
}

function formatRoles(text) {
  if (!selectedButton) return text;
  const data = topics[selectedButton.textContent];
  return boldRoles(text, data.roles);
}

function updateFlashcard() { 
  document.getElementById("cardElement").classList.remove("flipped"); 
  if(currentKeywords && currentKeywords.length > 0) { 
    const currentItem = currentKeywords[currentCardIndex]; 
    document.getElementById("cardFrontText").textContent = currentItem.word;
    document.getElementById("cardBackText").textContent = currentItem.def; 
    document.getElementById("cardTracker").textContent = `${currentCardIndex + 1} / ${currentKeywords.length}`; 
  } 
} 

function toggleCard() {
  document.getElementById("cardElement").classList.toggle("flipped");
}

function prevCard(event) { 
  event.stopPropagation();
  if (currentCardIndex > 0) { 
    currentCardIndex--; 
    updateFlashcard(); 
  } 
} 

function nextCard(event) { 
  event.stopPropagation();
  if (currentCardIndex < currentKeywords.length - 1) { 
    currentCardIndex++; 
    updateFlashcard(); 
  } 
} 

function resetConversation() {
  conversationIndices = [];
  renderPool();
  document.getElementById("conversationBox").innerHTML = `
    <p style="color:var(--text-muted); font-style:italic; text-align:center; margin-top:140px; font-size:15px;">
      Click the sentences from the left panel in the correct order to place them here.
    </p>`;
}

function renderPool() {
  const pool = document.getElementById("phrasePool");
  pool.innerHTML = "";
  workingPool.forEach((item, index) => {
    const btn = document.createElement("button");
    btn.className = "phrase-btn";
    btn.innerHTML = formatRoles(item.text);
    btn.onclick = () => pickLine(index);
    btn.id = "pool-btn-" + index;
    pool.appendChild(btn);
  });
}

function pickLine(poolIndex) {
  const item = workingPool[poolIndex];
  conversationIndices.push(item);
  
  const btn = document.getElementById("pool-btn-" + poolIndex);
  btn.disabled = true;
  btn.classList.add("used");

  renderConversationLines();
}

function renderConversationLines() {
  const box = document.getElementById("conversationBox");
  if (conversationIndices.length === 0) {
    resetConversation();
    return;
  }
  
  box.innerHTML = conversationIndices.map((item, idx) => `
    <div class="line-item" style="position:relative;">
      <span style="position:absolute; top:2px; right:6px; font-size:10px; font-weight:600; color:var(--text-muted);">#${idx + 1}</span>
      ${formatRoles(item.text)}
    </div>
  `).join("");
}

function checkConversation() {
  const currentLines = topics[selectedButton.textContent].dialogue;
  const box = document.getElementById("conversationBox");
  
  if(conversationIndices.length !== currentLines.length) {
    box.innerHTML = `<p style="color:var(--error); font-weight:800; font-size:16px; margin-bottom:10px;">⚠️ Please add all the conversation lines first.</p>` + box.innerHTML;
    return;
  }

  let allCorrect = true;
  let feedback = "";

  for(let i=0; i < currentLines.length; i++) {
    if(conversationIndices[i].originalIndex === i) {
      feedback += `<p style="color: var(--success); margin: 4px 0; font-size: 15px;">✔ ${formatRoles(conversationIndices[i].text)}</p>`;
    } else {
      allCorrect = false;
      feedback += `<p style="color: var(--error); margin: 4px 0; font-size: 15px;">✘ ${formatRoles(conversationIndices[i].text)}</p>`;
    }
  }

  box.innerHTML = allCorrect
    ? `<p style="color:var(--success); font-weight:800; font-size:17px; margin-bottom:10px;">🎉 Great Job! Everything is in the perfect order.</p>${feedback}`
    : `<p style="color:var(--error); font-weight:800; font-size:17px; margin-bottom:10px;">Some lines are incorrect. Click 'Clear Options' to try again.</p>${feedback}`;
}

function showCorrectOrder() {
  const currentLines = topics[selectedButton.textContent].dialogue;
  const box = document.getElementById("conversationBox");
  const data = topics[selectedButton.textContent];
  box.innerHTML = `
    <p style="color:var(--secondary-700); font-weight:800; font-size:16px; margin-bottom:10px;">Correct Answer Key:</p>
    ${currentLines.map(s => `<div class="line-item" style="margin-bottom:6px;">• ${boldRoles(s, data.roles)}</div>`).join("")}
  `;
}
